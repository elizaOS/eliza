  static void publication() {
    PermittedMediaCatalog valid = new PermittedMediaCatalog();
    ReadBridge bridge = read(valid);
    PluginCall success = new PluginCall();
    int[] callbacks = {0}, reads = {0};
    bridge.submitRead(success, () -> {
      reads[0]++;
      return new JSObject().value("id", "photo");
    }, () -> callbacks[0]++);
    check(reads[0] == 0 && success.resolved == 0, "admission queues the read");
    bridge.worker.drain();
    check(reads[0] == 1 && success.resolved == 0 && callbacks[0] == 0, "publication waits for main");
    valid.main.drain();
    check(success.resolved == 1 && success.rejected == 0 && callbacks[0] == 1,
        "current ticket publishes and reports success once");

    for (String change : List.of("epoch", "scope", "pause-resume", "lock", "profile", "close")) {
      PermittedMediaCatalog catalog = new PermittedMediaCatalog();
      ReadBridge host = read(catalog);
      PluginCall call = new PluginCall();
      int[] published = {0};
      host.submitRead(call, JSObject::new, () -> published[0]++);
      host.worker.drain();
      switch (change) {
        case "epoch" -> catalog.invalidate();
        case "scope" -> catalog.access = "selected";
        case "pause-resume" -> { catalog.foreground(false); catalog.foreground(true); }
        case "lock" -> catalog.context.keys.locked = true;
        case "profile" -> catalog.context.users.foreground = false;
        case "close" -> catalog.close();
      }
      catalog.main.drain();
      refused(call, change + " retires publication");
      check(published[0] == 0, change + " cannot report a published result");
    }

    PermittedMediaCatalog queued = new PermittedMediaCatalog();
    ReadBridge queuedHost = read(queued);
    PluginCall stale = new PluginCall();
    int[] io = {0};
    queuedHost.submitRead(stale, () -> { io[0]++; return new JSObject(); });
    queued.invalidate();
    queuedHost.worker.drain();
    refused(stale, "a stale queued read fails before I/O");
    check(io[0] == 0, "stale queued work never reads");

    PermittedMediaCatalog during = new PermittedMediaCatalog();
    ReadBridge duringHost = read(during);
    PluginCall changing = new PluginCall();
    duringHost.submitRead(changing, () -> {
      during.invalidate();
      return new JSObject();
    });
    duringHost.worker.drain();
    during.main.drain();
    refused(changing, "change during I/O retires its publication");

    PermittedMediaCatalog newest = new PermittedMediaCatalog();
    ReadBridge newestHost = read(newest);
    PluginCall old = new PluginCall(), fresh = new PluginCall();
    newestHost.submitRead(old, JSObject::new);
    newestHost.worker.drain();
    newest.access = "selected";
    newestHost.submitRead(fresh, JSObject::new);
    newestHost.worker.drain();
    newest.main.drain();
    refused(old, "a new admission never refreshes an old ticket");
    check(fresh.resolved == 1 && fresh.rejected == 0, "the new ticket still publishes");

    for (boolean nullValue : List.of(false, true)) {
      PermittedMediaCatalog failed = new PermittedMediaCatalog();
      ReadBridge failedHost = read(failed);
      PluginCall call = new PluginCall();
      int[] reported = {0};
      failedHost.submitRead(call, () -> {
        if (nullValue) { call.reject("read failed"); return null; }
        throw new IllegalStateException("read failed");
      }, () -> reported[0]++);
      failedHost.worker.drain();
      failed.main.drain();
      refused(call, "failed read settles once");
      check(reported[0] == 0, "failed read never reports publication");
    }

    PermittedMediaCatalog destroyed = new PermittedMediaCatalog();
    ReadBridge destroyedHost = read(destroyed);
    PluginCall gone = new PluginCall();
    destroyedHost.submitRead(gone, JSObject::new);

  static void selections() {
    SystemClock.now = 0;
    PermittedMediaCatalog catalog = new PermittedMediaCatalog();
    catalog.versions.put("one", "r0"); catalog.versions.put("two", "r0"); catalog.versions.put("other", "r1");
    Bridge bridge = new Bridge();
    Queue worker = new Queue();
    PermittedMediaReview review = new PermittedMediaReview(bridge, catalog, worker);
    List<String> ids = new ArrayList<>(List.of("one", "two"));
    List<String> revisions = new ArrayList<>(List.of("r0", "r0"));
    PluginCall first = new PluginCall(), busy = new PluginCall(), second = new PluginCall();
    review.request(first, ids, revisions, false, true, true, false);
    review.request(busy, List.of("other"), List.of("r1"), false, true, true, false);
    check("LIBRARY_BUSY".equals(busy.code), "one native review owns admission");
    ids.set(0, "other"); revisions.set(0, "r1");
    bridge.answer(Activity.RESULT_OK);
    check(worker.pending.isEmpty(), "accepted review waits for resumed activity");

    // A later review can occupy the pending slot while the first result waits
    // for resume. Its ids and revisions cannot replace the captured first request.
    review.request(second, List.of("other"), List.of("r1"), false, true, true, false);
    catalog.versions.put("one", "r1");
    bridge.resume();
    worker.drain();
    check(catalog.changed.equals(List.of("two@r0")), "only unchanged captured selections apply");
    check("partial".equals(first.value.getString("status")), "stale item stays an explicit partial result");
    check(second.resolved == 0, "the waiting first request never resolves a newer review");
    bridge.answer(Activity.RESULT_OK);
    worker.drain();
    check(catalog.changed.equals(List.of("two@r0", "other@r1")), "the second review owns its own selection");
    check("complete".equals(second.value.getString("status")), "the second review settles separately");
    bridge.resume(); worker.drain();
    check(catalog.changed.size() == 2, "another resume never repeats a dispatched review");

    for (String ending : List.of("close", "destroy", "expire", "decline")) {
      Bridge host = new Bridge();
      Queue io = new Queue();
      PermittedMediaReview held = new PermittedMediaReview(host, catalog, io);
      PluginCall call = new PluginCall();
      SystemClock.now = 0;
      held.request(call, List.of("two"), List.of("r0"), false, true, true, false);
      host.answer(ending.equals("decline") ? 0 : Activity.RESULT_OK);
      switch (ending) {
        case "close" -> { held.close(); host.resume(); }
        case "destroy" -> host.destroy();
        case "expire" -> { SystemClock.now = 120001; host.resume(); }
        case "decline" -> host.resume();
      }
      io.drain();
      refused(call, ending + " ends pending resume authority");
      check(io.pending.isEmpty() && catalog.changed.size() == 2, ending + " produces no write");
    }
    SystemClock.now = 0;
    Bridge stopping = new Bridge();
    Queue io = new Queue();
    PermittedMediaReview stopped = new PermittedMediaReview(stopping, catalog, io);
    PluginCall call = new PluginCall();
    stopping.resume();
    stopped.request(call, List.of("two"), List.of("r0"), false, true, true, false);
    stopping.answer(Activity.RESULT_OK);
    stopped.close();
    io.drain();
    check(catalog.changed.size() == 2 && "partial".equals(call.value.getString("status")),
        "closing after queueing reports an unverified selection without a write");
    PermittedMediaCatalog separate = new PermittedMediaCatalog();
    separate.versions.put("one", "r0"); separate.versions.put("other", "r1");
    Bridge returning = new Bridge();
    Queue writes = new Queue();
    PermittedMediaReview two = new PermittedMediaReview(returning, separate, writes);
    PluginCall a = new PluginCall(), b = new PluginCall();
    two.request(a, List.of("one"), List.of("r0"), false, true, true, false);
    returning.answer(Activity.RESULT_OK);
    two.request(b, List.of("other"), List.of("r1"), false, true, true, false);
    returning.answer(Activity.RESULT_OK);
    check(writes.pending.isEmpty(), "two accepted requests wait for their resume callbacks");
    returning.resume();
    check(writes.pending.size() == 2, "resume synchronously dispatches each captured observer");
    writes.drain();
    check(separate.changed.equals(List.of("one@r0", "other@r1")),
        "two resume observers retain their separate captured selections");
    check(a.resolved == 1 && b.resolved == 1, "each resumed request settles once");
    System.out.println("PASS Photos reviewed selection ownership (9 scenarios)");
  }
