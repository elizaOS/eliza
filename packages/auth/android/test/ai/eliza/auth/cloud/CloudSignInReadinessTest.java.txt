 final Map<String, CloudSignInReadiness.Result> readiness = new HashMap<>(), stored = new HashMap<>();

 static class Main { final List<Runnable> delayed = new ArrayList<>(); final List<Long> delays = new ArrayList<>(); void postDelayed(Runnable task, long delay) { delayed.add(task); delays.add(delay); } void next() { delays.remove(0); delayed.remove(0).run(); } }

 final List<JSObject> changes = new ArrayList<>();
 void notifyListeners(String event, JSObject value, boolean retained) { check("signInReadinessChanged".equals(event) && !retained, "only an unretained readiness change is announced"); changes.add(value); }
 boolean browserMode, appLink, assetLinks;

 // Measurement ports.
 boolean googleNative = true, registered = true, email = true, captcha = false; String route = "present"; int measures; boolean measureFails; CloudSignInReadiness.Result next;
 final List<String> reads = new ArrayList<>();
 Response get(String url) throws Exception {
  reads.add(url);
  if (url.contains("/mobile/config?")) return new Response(registered ? 200 : 503, new JSONObject().put("success", true).put("clientId", "ai.example.signin").put("environment", "production").put("redirectUri", "https://eliza.app/example/auth/callback").put("codeChallengeMethod", "S256"), null);
  if (url.contains("/steward/auth/providers?")) return new Response(200, new JSONObject().put("email", email).put("sms", true).put("googleNative", googleNative).put("google", true), null);
  if (url.contains("/steward/auth/oauth/google/authorize?")) return new Response(302, new JSONObject(), "https://accounts.google.com/o/oauth2/v2/auth");
  throw new AssertionError("unexpected read " + url);
 }

  if (!CloudEmailCode.SEND_PATH.equals(path)) throw new AssertionError(path);

 static void check(boolean value, String why) { if (!value) throw new AssertionError(why); }
 static PluginCall call(boolean refresh) { PluginCall call = new PluginCall(); call.refresh = refresh; return call; }
 static boolean ready(PluginCall call) { return Boolean.TRUE.equals(call.result.values.get("inApp")); }
 /** Probe ports for status/probe that delegate to the measurement stub or a scripted result. */
 static CloudSignInReadinessTest scripted() {
  return new CloudSignInReadinessTest() { @Override CloudSignInReadiness.Result measure(CloudSignInEnvironment environment) throws Exception { measures++; if (measureFails) throw new Exception("Synthetic offline"); return next; } };
 }
 public static void main(String[] args) throws Exception {
  CloudSignInEnvironment e = TestRegistration.PRODUCTION;
  // measure(): each method independent; a missing registration disables code methods.
  CloudSignInReadinessTest m = new CloudSignInReadinessTest();
  CloudSignInReadiness.Result all = m.measure(e);
  check(all.email && all.google && all.phone && all.until == System.now + CloudSignInReadiness.READY_TTL_MS, "all methods ready");
  check(m.reads.get(0).equals("https://api.eliza.app/api/v1/app-auth/mobile/config?clientId=ai.example.signin&environment=production&redirectUri=https%3A%2F%2Feliza.app%2Fexample%2Fauth%2Fcallback"), "exact registration read");
  m.googleNative = false; CloudSignInReadiness.Result noGoogle = m.measure(e); check(noGoogle.email && !noGoogle.google, "Google off keeps email");
  m.googleNative = true; m.route = "timeout"; CloudSignInReadiness.Result slowRoute = m.measure(e); check(!slowRoute.email && slowRoute.google && slowRoute.phone, "an email route timeout keeps Google and SMS");
  m.route = "absent"; check(!m.measure(e).email, "absent email route");
  m.route = "present"; m.email = false; CloudSignInReadiness.Result noEmail = m.measure(e); check(!noEmail.email && noEmail.google, "email not offered");
  m.email = true; m.registered = false; CloudSignInReadiness.Result unregistered = m.measure(e); check(!unregistered.inApp && unregistered.until == System.now + CloudSignInReadiness.NOT_READY_TTL_MS, "unregistered app");

  CloudSignInReadinessTest browser = new CloudSignInReadinessTest();
  browser.browserMode = true;
  check(!browser.measure(e).google, "browser mode requires verified app link");
  browser.appLink = true;
  check(!browser.measure(e).google, "browser mode requires exact Asset Links grant");
  browser.assetLinks = true;
  check(browser.measure(e).google, "verified link, supported browser and provider redirect admit browser Google");

  // status/probe: fresh caches are served, an explicit retry refreshes only negative or partial answers.
  for (boolean persisted : new boolean[]{false, true}) {
   CloudSignInReadinessTest h = scripted(); CloudSignInReadiness.Result old = CloudSignInReadiness.Result.notReady(System.now);
   (persisted ? h.stored : h.readiness).put(e.name, old);
   PluginCall passive = call(false); h.status(passive);
   check(passive.resolved == 1 && !ready(passive) && h.prober.tasks.isEmpty(), "ordinary lookup keeps a fresh negative cache");
   PluginCall retry = call(true), joined = call(true); h.status(retry); h.status(joined);
   check(h.prober.tasks.size() == 1 && retry.resolved == 0 && joined.resolved == 0, "an explicit retry queues exactly one probe and joins waiters");
   h.next = new CloudSignInReadiness.Result(true, false, false, System.now + CloudSignInReadiness.NOT_READY_TTL_MS); h.prober.drain();
   check(h.measures == 1 && retry.resolved == 1 && joined.resolved == 1 && ready(retry) && Boolean.FALSE.equals(retry.result.values.get("google")), "fresh measurement answers every waiter");
   check(h.stored.get(e.name) == h.next && h.probing.isEmpty(), "cache persisted and in-flight released");
  }
  CloudSignInReadinessTest positive = scripted(); positive.readiness.put(e.name, CloudSignInReadiness.Result.measured(true, true, true, System.now));
  PluginCall hot = call(true); positive.status(hot); check(hot.resolved == 1 && positive.measures == 0, "a complete cache survives an explicit retry");
  System.now += CloudSignInReadiness.READY_TTL_MS - 1; PluginCall later = call(false); positive.status(later); positive.handleOnResume();
  check(later.resolved == 1 && positive.measures == 0 && positive.prober.tasks.isEmpty(), "a complete cache is served for its whole six hours, also across resume");
  System.now = 1_000_000;

  // Per-method TTL: Google not ready (the return link was unverified at first launch) while email and SMS are ready.
  long start = System.now;
  CloudSignInReadinessTest fresh = scripted(); fresh.next = CloudSignInReadiness.Result.measured(true, false, true, System.now);
  fresh.probe(e, null, false); fresh.prober.drain();
  check(fresh.measures == 1 && fresh.changes.isEmpty() && fresh.stored.get(e.name).until == start + CloudSignInReadiness.READY_TTL_MS, "the first-launch probe caches the ready methods");
  PluginCall soon = call(false); fresh.status(soon);
  check(soon.resolved == 1 && Boolean.FALSE.equals(soon.result.values.get("google")) && fresh.prober.tasks.isEmpty(), "inside the short TTL the cached answer is served");
  System.now = start + CloudSignInReadiness.METHOD_RECHECK_MS;
  PluginCall opened = call(false), joinedOpen = call(false); fresh.status(opened); fresh.status(joinedOpen);
  check(opened.resolved == 1 && joinedOpen.resolved == 1 && Boolean.TRUE.equals(opened.result.values.get("email")) && Boolean.FALSE.equals(opened.result.values.get("google")), "after the short TTL the cached ready methods are still served at once: opening sign-in never waits for a recheck");
  check(fresh.prober.tasks.size() == 1, "and the missing method is measured again behind that answer, once, so Google=false is never kept for six hours");
  fresh.next = CloudSignInReadiness.Result.measured(true, true, true, System.now); fresh.prober.drain();
  check(fresh.measures == 2 && opened.resolved == 1 && joinedOpen.resolved == 1, "the calls were answered once");
  check(fresh.changes.size() == 1 && Boolean.TRUE.equals(fresh.changes.get(0).values.get("google")) && "production".equals(fresh.changes.get(0).values.get("environment")), "Google appears without an explicit retry: the change is announced once with its deployment");
  PluginCall after = call(false); fresh.status(after); check(Boolean.TRUE.equals(after.result.values.get("google")) && fresh.prober.tasks.isEmpty(), "the next lookup has it");
  for (boolean[] row : new boolean[][]{{false, true, true}, {true, true, false}}) {
   System.now = start;
   CloudSignInReadinessTest other = scripted(); other.readiness.put(e.name, CloudSignInReadiness.Result.measured(row[0], row[1], row[2], System.now));
   System.now += CloudSignInReadiness.METHOD_RECHECK_MS; PluginCall due = call(false); other.status(due);
   check(due.resolved == 1 && other.prober.tasks.size() == 1, "email and SMS are measured again on the short TTL as well");
   other.next = CloudSignInReadiness.Result.measured(row[0], row[1], row[2], System.now); other.prober.drain();
   check(due.resolved == 1 && other.changes.isEmpty(), "an unchanged answer announces nothing");
  }
  // An explicit Retry of an incomplete answer is the one lookup that waits for the measurement.
  System.now = start;
  CloudSignInReadinessTest asked = scripted(); asked.readiness.put(e.name, CloudSignInReadiness.Result.measured(true, false, true, System.now));
  PluginCall explicit = call(true); asked.status(explicit);
  check(explicit.resolved == 0 && asked.prober.tasks.size() == 1, "Retry waits for a fresh measurement");
  asked.next = CloudSignInReadiness.Result.measured(true, true, true, System.now); asked.prober.drain();
  check(explicit.resolved == 1 && Boolean.TRUE.equals(explicit.result.values.get("google")), "and is answered by it");

  // A recheck that gets no result (a few seconds without a connection) keeps the cached ready methods.
  System.now = start;
  CloudSignInReadinessTest gap = scripted(); CloudSignInReadiness.Result partialCache = CloudSignInReadiness.Result.measured(true, false, true, System.now);
  gap.readiness.put(e.name, partialCache); gap.stored.put(e.name, partialCache);
  System.now += CloudSignInReadiness.METHOD_RECHECK_MS; gap.measureFails = true;
  PluginCall open = call(false); gap.status(open); gap.prober.drain();
  check(gap.measures == 1 && open.resolved == 1 && Boolean.TRUE.equals(open.result.values.get("email")) && Boolean.TRUE.equals(open.result.values.get("phone")), "the open sign-in screen keeps email and SMS");
  CloudSignInReadiness.Result keptCache = gap.readiness.get(e.name);
  check(keptCache.email && keptCache.phone && !keptCache.google && keptCache.methodsKnown && keptCache.until == partialCache.until && gap.stored.get(e.name) == keptCache, "a failed recheck neither replaces nor persists the answer as not ready");
  check(gap.changes.isEmpty(), "and announces nothing");
  check(keptCache.recheckAt == System.now + CloudSignInReadiness.METHOD_RECHECK_MS && !keptCache.recheckDue(System.now + CloudSignInReadiness.METHOD_RECHECK_MS - 1) && keptCache.recheckDue(System.now + CloudSignInReadiness.METHOD_RECHECK_MS), "the next recheck is due in 30 seconds");
  PluginCall retryGap = call(true); gap.status(retryGap); gap.prober.drain();
  check(retryGap.resolved == 1 && Boolean.TRUE.equals(retryGap.result.values.get("email")) && gap.changes.isEmpty(), "an explicit Retry without a connection also keeps what was ready");
  gap.measureFails = false; gap.next = CloudSignInReadiness.Result.measured(true, true, true, System.now);
  System.now += CloudSignInReadiness.METHOD_RECHECK_MS; gap.status(call(false)); gap.prober.drain();
  check(gap.changes.size() == 1 && gap.readiness.get(e.name).complete(), "once the connection is back the missing method is measured and announced");
  // Only a cache that is no longer fresh falls back to not ready on a failed measurement.
  System.now = start;
  CloudSignInReadinessTest expired = scripted(); expired.readiness.put(e.name, CloudSignInReadiness.Result.measured(true, false, true, System.now));
  System.now += CloudSignInReadiness.READY_TTL_MS; expired.measureFails = true;
  PluginCall cold = call(false); expired.status(cold);
  check(cold.resolved == 0, "an expired answer is not served");
  expired.prober.drain();
  check(cold.resolved == 1 && !ready(cold) && !expired.readiness.get(e.name).methodsKnown, "with no fresh answer a failed measurement is not ready");

  // Resume: a due method is measured in the background while a sign-in screen asked recently; an attempt
  // whose page is in the browser's own task (a Home host) is left alone.
  System.now = start;
  CloudSignInReadinessTest resumed = scripted(); resumed.readiness.put(e.name, CloudSignInReadiness.Result.measured(true, false, true, System.now));
  resumed.attempt = new Attempt(); resumed.attempt.google = true; resumed.attempt.browserLaunched = true; Attempt waitingInBrowser = resumed.attempt;
  resumed.handleOnPause(); check(!waitingInBrowser.browserPaused, "Home on a launcher host does not mark the page closed");
  resumed.handleOnResume(); check(resumed.prober.tasks.isEmpty(), "resume inside the short TTL sends nothing");
  System.now += CloudSignInReadiness.METHOD_RECHECK_MS; resumed.handleOnResume(); resumed.handleOnResume();
  check(resumed.prober.tasks.size() == 1, "resume measures a due method once");
  resumed.next = CloudSignInReadiness.Result.measured(true, true, true, System.now); resumed.prober.drain();
  check(resumed.changes.size() == 1 && resumed.attempt == waitingInBrowser && waitingInBrowser.ended == null && resumed.main.delayed.isEmpty(), "resume announces the change and never ends or times a browser attempt in the browser's own task");
  PluginCall during = call(false); resumed.status(during);
  check(Boolean.TRUE.equals(during.result.values.get("active")) && "google".equals(during.result.values.get("kind")) && !during.result.values.containsKey("activationPending"), "the waiting attempt is still reported active");
  resumed.activating.add(e.name); PluginCall pendingKey = call(false); resumed.status(pendingKey);
  check(Boolean.TRUE.equals(pendingKey.result.values.get("activationPending")), "a stored key that waits for its acknowledgement is reported");

  // Any other host: the page is in the host's own task, so coming back without a return means it was closed.
  for (String how : new String[]{"closed", "returned", "hinted", "cancelled"}) {
   CloudSignInReadinessTest hostTask = scripted(); hostTask.readiness.put(e.name, CloudSignInReadiness.Result.measured(true, true, true, System.now));
   Attempt tab = new Attempt(); tab.google = true; tab.browserLaunched = true; tab.browserInHostTask = true; hostTask.attempt = tab;
   hostTask.handleOnPause(); check(tab.browserPaused, "the page covers the host");
   hostTask.handleOnResume();
   check(tab.ended == null && hostTask.main.delays.equals(List.of(BROWSER_RETURN_GRACE_MS)), "a return that is on its way gets a short grace");
   if (how.equals("returned")) tab.returnClaimed = true; if (how.equals("hinted")) tab.returnHint = true; if (how.equals("cancelled")) { tab.cancelled = true; hostTask.attempt = null; }
   hostTask.main.next();
   if (how.equals("closed")) check("browser-closed".equals(tab.ended) && hostTask.attempt == null, "a closed tab ends the attempt, so email and SMS are not refused as login-active for ten minutes");
   else check(tab.ended == null, "a return, a hint or a cancel decides instead: " + how);
  }
  CloudSignInReadinessTest notLaunched = scripted(); notLaunched.readiness.put(e.name, CloudSignInReadiness.Result.measured(true, true, true, System.now));
  notLaunched.attempt = new Attempt(); notLaunched.attempt.browserInHostTask = true;
  notLaunched.handleOnPause(); notLaunched.handleOnResume();
  check(notLaunched.main.delayed.isEmpty() && notLaunched.attempt.ended == null, "an attempt without a launched page (email, SMS, the account picker) is never ended by resume");

  // Resume is bounded: a launcher resumes on every Home press. With one method that never becomes ready
  // (a tenant without SMS) nothing is measured unless a sign-in screen asked recently, and then on a growing wait.
  System.now = start;
  CloudSignInReadinessTest launcher = scripted(); launcher.next = CloudSignInReadiness.Result.measured(true, true, false, System.now);
  launcher.probe(e, null, false, true); launcher.prober.drain();
  check(launcher.measures == 1, "the load-time warm-up measures once");
  for (int minute = 1; minute <= 24 * 60; minute++) { System.now = start + minute * 60_000L; launcher.next = CloudSignInReadiness.Result.measured(true, true, false, System.now); launcher.handleOnResume(); launcher.prober.drain(); }
  check(launcher.measures == 1 && launcher.changes.isEmpty(), "a day of resumes with no sign-in screen sends no probe at all");
  check(launcher.renewalChecks == 24 * 60, "the renewal check still runs (it limits itself)");
  System.now = start;
  CloudSignInReadinessTest screen = scripted(); screen.next = CloudSignInReadiness.Result.measured(true, true, false, System.now);
  PluginCall mounted = call(false); screen.status(mounted); screen.prober.drain();
  check(screen.measures == 1 && mounted.resolved == 1, "the sign-in screen asks once");
  List<Long> probedAt = new ArrayList<>();
  for (int tick = 1; tick <= 120; tick++) {
   System.now = start + tick * 30_000L;
   if (tick % 10 == 0) screen.statusAskedAt = System.now;
   screen.next = CloudSignInReadiness.Result.measured(true, true, false, System.now);
   int before = screen.measures; screen.handleOnResume(); screen.prober.drain();
   if (screen.measures > before) probedAt.add((System.now - start) / 1000);
  }
  check(probedAt.equals(List.of(30L, 150L, 750L)), "an hour of resumes every 30 seconds with the sign-in screen open sends three probes on a growing wait (30 s, 2 min, 10 min, then hourly), not 120: " + probedAt);
  check(screen.changes.isEmpty() && screen.readiness.get(e.name).unchanged == 3, "an unchanged answer announces nothing and keeps counting");
  PluginCall reopened = call(false); screen.status(reopened);
  check(reopened.resolved == 1 && screen.prober.tasks.size() == 1, "the sign-in screen asking again still rechecks on the short TTL, behind the cached answer");
  screen.next = CloudSignInReadiness.Result.measured(true, true, true, System.now); screen.prober.drain();
  check(screen.changes.size() == 1 && screen.readiness.get(e.name).unchanged == 0, "a changed answer is announced and the wait starts over");
  System.now = start;
  CloudSignInReadinessTest idleScreen = scripted(); idleScreen.next = CloudSignInReadiness.Result.measured(true, true, false, System.now);
  idleScreen.status(call(false)); idleScreen.prober.drain();
  System.now += CloudSignInReadiness.RESUME_PROBE_WINDOW_MS; idleScreen.handleOnResume();
  check(idleScreen.prober.tasks.isEmpty(), "ten minutes after the last lookup a resume measures nothing");

  // Fresh install in browser mode: Android verifies the return link after the first probe.
  System.now = start;
  CloudSignInReadinessTest install = new CloudSignInReadinessTest(); install.browserMode = true;
  install.probe(e, null, false); install.prober.drain();
  check(!install.readiness.get(e.name).google && install.readiness.get(e.name).email && install.linkWatch.contains(e.name) && install.main.delays.equals(List.of(CloudSignInReadiness.LINK_WATCH_MS[0])), "an unverified return link is watched");
  install.probe(e, null, true); install.prober.drain();
  check(install.main.delayed.size() == 1, "one watch per deployment");
  int requests = install.reads.size();
  install.main.next(); install.prober.drain();
  check(install.reads.size() == requests && install.main.delays.equals(List.of(CloudSignInReadiness.LINK_WATCH_MS[1])) && install.changes.isEmpty(), "a pending link costs no request");
  install.appLink = true; install.assetLinks = true;
  install.main.next(); install.prober.drain();
  check(install.readiness.get(e.name).google && install.main.delayed.isEmpty() && !install.linkWatch.contains(e.name), "the verified link is measured at once and the watch ends");
  check(install.changes.size() == 1 && Boolean.TRUE.equals(install.changes.get(0).values.get("google")), "Google is announced to the open sign-in screen");
  CloudSignInReadinessTest never = new CloudSignInReadinessTest(); never.browserMode = true;
  never.probe(e, null, false); never.prober.drain();
  for (int round = 0; round < CloudSignInReadiness.LINK_WATCH_MS.length; round++) { never.main.next(); never.prober.drain(); }
  check(never.main.delayed.isEmpty() && !never.linkWatch.contains(e.name) && never.changes.isEmpty(), "the watch is bounded when the link never verifies");
  CloudSignInReadinessTest gone = new CloudSignInReadinessTest(); gone.browserMode = true;
  gone.probe(e, null, false); gone.prober.drain(); gone.destroyed = true; gone.appLink = true; gone.main.next(); gone.prober.drain();
  check(gone.main.delayed.isEmpty() && !gone.readiness.get(e.name).google, "a closed plugin stops watching");
  System.now = 1_000_000;
  CloudSignInReadinessTest partial = scripted(); partial.readiness.put(e.name, new CloudSignInReadiness.Result(true, false, false, System.now + CloudSignInReadiness.NOT_READY_TTL_MS));
  PluginCall partialRetry = call(true); partial.status(partialRetry);
  check(partial.prober.tasks.size() == 1 && partialRetry.resolved == 0, "partial readiness is measured again on an explicit retry");
  partial.next = new CloudSignInReadiness.Result(true, true, false, System.now + CloudSignInReadiness.READY_TTL_MS); partial.prober.drain();
  check(partial.measures == 1 && Boolean.TRUE.equals(partialRetry.result.values.get("google")), "Google is enabled only by a fresh full measurement");
  CloudSignInReadinessTest warming = scripted(); warming.probe(e, null, false);
  PluginCall duringWarm = call(true); warming.status(duringWarm);
  check(warming.prober.tasks.size() == 1 && duringWarm.resolved == 0, "an explicit retry joins the warm-up probe");
  warming.next = CloudSignInReadiness.Result.notReady(System.now); warming.prober.drain();
  check(warming.measures == 1 && duringWarm.resolved == 1 && !ready(duringWarm) && warming.prober.tasks.isEmpty(), "a failed warm-up answers its waiters once and never retries itself");
  CloudSignInReadinessTest failed = scripted(); failed.readiness.put(e.name, CloudSignInReadiness.Result.notReady(System.now)); failed.measureFails = true;
  PluginCall retry = call(true); failed.status(retry); failed.prober.drain();
  check(retry.resolved == 1 && !ready(retry) && failed.measures == 1 && failed.prober.tasks.isEmpty() && failed.readiness.get(e.name).until == System.now + CloudSignInReadiness.NOT_READY_TTL_MS, "a failed measurement stays negative and never retries itself");
  CloudSignInReadinessTest closed = scripted(); closed.readiness.put(e.name, CloudSignInReadiness.Result.notReady(System.now)); closed.prober.closed = true;
  PluginCall refused = call(true); closed.status(refused); check(refused.resolved == 1 && !ready(refused) && closed.probing.isEmpty(), "executor refusal releases waiters fail closed");
