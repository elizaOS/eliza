package ai.elizaos.app;

import ai.eliza.plugins.agent.runtime.CredentialShutdownLease;
import ai.eliza.plugins.securestore.nativeonly.CredentialIntentFence;
import ai.eliza.plugins.securestore.nativeonly.CredentialMutationConductor;
import ai.eliza.plugins.securestore.nativeonly.CredentialSlotMutation;
import java.util.ArrayDeque;
import java.util.HashMap;
import java.util.Map;

/** Runs the actual native mutation classes and the app's lifecycle binding. */
public final class ResidentCredentialMutationTest {
    /* ACTUAL_LIFECYCLE */
    /* ACTUAL_RESERVE */

    private static Fixture fixture;
    private static final class Context {}
    private static final class JsonCredentialSlots {
        static final Object LOCK = new Object();
    }
    private static final class SystemClock {
        static long elapsedRealtime() { return fixture.now; }
    }
    // Advances the fixture's clock between actual conductor polls.
    private static final class Thread {
        static void sleep(long millis) {
            fixture.now += millis;
            if (!fixture.onSleep.isEmpty()) fixture.onSleep.removeFirst().run();
        }
    }

    private static final class ElizaCredentialRetirement {
        static long reserve() { return fixture.lease.acquire(); }
        static void request(Context context, long request) {
            fixture.lastRequest = fixture.lease.retry(request);
            fixture.stopRequests++;
            if (fixture.failDispatch) throw new IllegalStateException("controlled dispatch failure");
            if (fixture.confirmOnDispatch) {
                fixture.stopped = true;
                fixture.lease.acknowledge(request, true);
            }
        }
        static boolean stopped(Context context) { return fixture.stopped; }
        static boolean acknowledged(long request) { return fixture.lease.isConfirmed(request); }
        static void release(long request) { fixture.lease.release(request); }
        static void abandon(long request) { fixture.lease.abandon(request); }
    }

    private static final class Binding {
        final Context context = new Context();
        final Runnable guard;
        final Config config = new Config();
        final Cloud cloud = new Cloud();
        final NativeSlots nativeSlots = new NativeSlots();
        Binding() { this(() -> {}); }
        Binding(Runnable guard) { this.guard = guard; }
        String read(String name) throws Exception { return cloud.read(name); }
        /* ACTUAL_SLOT_KINDS */
        /* ACTUAL_COMPARE */
        /* ACTUAL_MUTATE */
        /* ACTUAL_WITH_LOCKS */
        /* ACTUAL_HOST */
    }

    private static final class CloudSignInEnvironment {}
    private static final class Config {
        final String pendingSlot = "pending";
        final CloudSignInEnvironment environment = new CloudSignInEnvironment();
        CloudSignInEnvironment[] environments() { return new CloudSignInEnvironment[]{ environment }; }
        String credentialSlot(CloudSignInEnvironment environment) { return "credential"; }
        String receiptSlot(CloudSignInEnvironment environment) { return "receipt"; }
        String retiringSlot(CloudSignInEnvironment environment) { return "retiring"; }
    }
    private static void locked() {
        check(java.lang.Thread.holdsLock(LIFECYCLE) && java.lang.Thread.holdsLock(JsonCredentialSlots.LOCK),
            "slot effect holds lifecycle and credential locks");
    }
    private static final class Cloud {
        String read(String slot) {
            locked();
            fixture.reads++;
            if (!"credential".equals(slot)) return fixture.nativeValues.get(slot);
            if (fixture.failReadback && fixture.writes > 0)
                throw new IllegalStateException("controlled lost readback");
            return fixture.value;
        }
        void write(String slot, String value) {
            locked();
            if (!"credential".equals(slot)) {
                fixture.nativeWrites++;
                fixture.nativeValues.put(slot, value);
                return;
            }
            fixture.writes++;
            if (!fixture.omitWrite) fixture.value = value;
        }
        void remove(String slot) {
            locked();
            if ("credential".equals(slot)) write(slot, null);
            else { fixture.nativeRemovals++; fixture.nativeValues.remove(slot); }
        }
    }
    private static final class NativeSlots {
        void remove(String slot) {
            locked();
            check("pending".equals(slot), "only pending browser state changes during credential mutation");
            fixture.changed++;
            fixture.nativeValues.remove(slot);
        }
    }
    private static final class Store {
        final Binding binding;
        Store(Runnable guard) { binding = new Binding(guard); }
        /* ACTUAL_CAS */
    }

    private static final class Fixture {
        final CredentialShutdownLease lease = new CredentialShutdownLease();
        final ArrayDeque<Runnable> onSleep = new ArrayDeque<>();
        final Map<String, String> nativeValues = new HashMap<>();
        String value = "original";
        boolean stopped, confirmOnDispatch, failDispatch, omitWrite, failReadback;
        long now, lastRequest;
        int stopRequests, writes, changed, reads, nativeWrites, nativeRemovals;

        void replace(Runnable owner, String replacement) throws Exception {
            CredentialSlotMutation.replace(new Binding().host(), owner, () -> {
                if (failReadback && writes > 0) throw new IllegalStateException("controlled lost readback");
                return value;
            }, replacement, () -> {
                writes++;
                if (!omitWrite) value = replacement;
                return null;
            }, () -> {
                changed++;
                return null;
            });
        }
    }

    private static Fixture fresh() {
        check(mutationEpoch == -1, "previous mutation epoch released");
        fixture = new Fixture();
        return fixture;
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private interface Work { void run() throws Exception; }

    private static void rejects(Work work, String message) {
        try {
            work.run();
        } catch (IllegalStateException expected) {
            check(expected.getMessage().contains(message), "unexpected refusal: " + expected.getMessage());
            return;
        } catch (Exception failure) {
            throw new AssertionError("unexpected exception", failure);
        }
        throw new AssertionError("expected refusal: " + message);
    }

    private static void unchangedKeepsRunning() throws Exception {
        Fixture f = fresh();
        f.replace(reserve(), "original");
        check(f.stopRequests == 0 && f.changed == 0 && f.writes == 1, "same value keeps the resident");
        f.lease.beforeStart();
    }

    private static void exactAbsenceAndReceipt() throws Exception {
        Fixture f = fresh();
        f.onSleep.add(() -> {
            f.lease.acknowledge(f.lastRequest, true);
            check(f.writes == 0 && !f.stopped, "receipt alone has no durable write");
        });
        f.onSleep.add(() -> {
            check(f.writes == 0, "unconfirmed absence still blocks");
            f.lease.acknowledge(f.lastRequest, false);
            f.stopped = true;
            f.lease.acknowledge(f.lastRequest + 1, true);
            rejects(f.lease::beforeStart, "Credential shutdown is pending");
        });
        f.onSleep.add(() -> {
            check(f.writes == 0, "foreign receipt does not admit a write");
            f.lease.acknowledge(f.lastRequest, true);
        });
        f.replace(reserve(), null);
        check(f.value == null && f.changed == 1 && f.writes == 1, "sign-out writes once after both checks");
        check(f.lease.isReleased(f.lastRequest), "confirmed readback releases exact lease");
        f.lease.beforeStart();
    }

    private static void reservationSupersedesQueuedWrite() throws Exception {
        Fixture f = fresh();
        Runnable earlier = reserve();
        Runnable current = reserve();
        rejects(() -> f.replace(earlier, "older"), "Credential intent changed");
        check(f.writes == 0 && f.stopRequests == 0, "stale queued owner has no effects");
        f.confirmOnDispatch = true;
        f.replace(current, "newer");
        check("newer".equals(f.value) && f.writes == 1, "current owner writes");
    }

    private static void olderCleanupKeepsNewerUnconfirmedLease() throws Exception {
        Fixture f = fresh();
        f.onSleep.add(() -> {
            Runnable newer = reserve();
            f.confirmOnDispatch = true;
            f.omitWrite = true;
            rejects(() -> f.replace(newer, "newer"), "Credential commit unconfirmed");
            check(f.lease.isConfirmed(f.lastRequest), "newer exact stop is confirmed");
        });
        rejects(() -> f.replace(reserve(), "older"), "Credential intent changed");
        check(f.stopRequests == 2 && f.writes == 1 && "original".equals(f.value), "only newer write attempted");
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
    }

    private static void preCommitFailureWaitsForExactReceipt(boolean dispatch) {
        Fixture f = fresh();
        f.failDispatch = dispatch;
        f.onSleep.add(() -> f.now = 15_000);
        rejects(() -> f.replace(reserve(), "newer"),
            dispatch ? "controlled dispatch failure" : "Native shutdown unconfirmed");
        check(f.writes == 0 && f.changed == 0 && "original".equals(f.value), "pre-commit failure has no write");
        f.lease.acknowledge(f.lastRequest + 1, true);
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
        f.lease.acknowledge(f.lastRequest, true);
        check(f.lease.isReleased(f.lastRequest), "late exact receipt releases an abandoned pre-commit stop");
        f.lease.beforeStart();
    }

    private static void writeWithLostReadbackKeepsFence() {
        Fixture f = fresh();
        f.confirmOnDispatch = true;
        f.failReadback = true;
        rejects(() -> f.replace(reserve(), "newer"), "controlled lost readback");
        check("newer".equals(f.value) && f.changed == 1 && f.writes == 1, "write may have happened");
        check(f.lease.isConfirmed(f.lastRequest) && !f.lease.isReleased(f.lastRequest), "unconfirmed write keeps exact stop");
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
    }

    private static void credentialCompareMismatchHasNoEffects() throws Exception {
        Fixture f = fresh();
        check(!new Store(reserve()).compareExchange("credential", "different", "successor"), "mismatch refuses CAS");
        check(f.stopRequests == 0 && f.writes == 0 && f.changed == 0 && "original".equals(f.value),
            "mismatch neither retires the resident nor changes storage");
        f.lease.beforeStart();
    }

    private static void staleCompareChecksOwnerBeforeReading() {
        Fixture f = fresh();
        Store stale = new Store(reserve());
        reserve();
        rejects(() -> stale.compareExchange("credential", "different", "successor"), "Credential intent changed");
        rejects(() -> stale.compareExchange("receipt", "different", null), "Credential intent changed");
        check(f.reads == 0 && f.stopRequests == 0 && f.writes == 0 && f.nativeRemovals == 0,
            "even mismatched CAS checks the owner before reading or changing any slot");
    }

    private static void credentialCompareWaitsForExactStop() throws Exception {
        Fixture f = fresh();
        f.onSleep.add(() -> {
            check(f.writes == 0, "CAS has not written before shutdown");
            f.stopped = true;
            f.lease.acknowledge(f.lastRequest + 1, true);
        });
        f.onSleep.add(() -> {
            check(f.writes == 0, "foreign stop receipt cannot admit credential CAS");
            f.lease.acknowledge(f.lastRequest, true);
        });
        check(new Store(reserve()).compareExchange("credential", "original", "successor"), "exact CAS succeeds");
        check("successor".equals(f.value) && f.writes == 1 && f.changed == 1 && f.stopRequests == 1,
            "credential CAS uses the actual guarded mutation and pending retirement");
        check(f.lease.isReleased(f.lastRequest), "readback releases only the exact stop lease");
        f.lease.beforeStart();
    }

    private static void sameCredentialCompareKeepsResident() throws Exception {
        Fixture f = fresh();
        check(new Store(reserve()).compareExchange("credential", "original", "original"), "idempotent CAS confirms");
        check(f.stopRequests == 0 && f.changed == 0 && f.writes == 1, "same credential needs no retirement");
        f.lease.beforeStart();
    }

    private static void credentialCompareRepairsEmptySelection() throws Exception {
        Fixture f = fresh();
        f.value = null;
        f.confirmOnDispatch = true;
        check(new Store(reserve()).compareExchange("credential", null, "successor"), "empty selected slot is repairable");
        check("successor".equals(f.value) && f.writes == 1 && f.changed == 1 && f.stopRequests == 1,
            "repair still crosses resident retirement and durable readback");
    }

    private static void credentialCompareLostReadbackKeepsFence() {
        Fixture f = fresh();
        f.confirmOnDispatch = true;
        f.failReadback = true;
        rejects(() -> new Store(reserve()).compareExchange("credential", "original", "successor"),
            "controlled lost readback");
        check("successor".equals(f.value) && f.writes == 1 && f.changed == 1, "CAS write may have committed");
        check(f.lease.isConfirmed(f.lastRequest) && !f.lease.isReleased(f.lastRequest),
            "unconfirmed CAS readback retains the exact shutdown fence");
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
    }

    private static void nativeComparePreservesNewerReceipts() throws Exception {
        Fixture f = fresh();
        f.nativeValues.put("receipt", "newer-proof");
        Store store = new Store(reserve());
        check(!store.compareExchange("receipt", "older-proof", null), "stale receipt cleanup refuses");
        check("newer-proof".equals(f.nativeValues.get("receipt")) && f.nativeRemovals == 0,
            "newer receipt stays intact");
        check(store.compareExchange("receipt", "newer-proof", "confirmed-proof"), "matching receipt advances");
        check(store.compareExchange("retiring", null, "exact-duty"), "retirement duty uses its existing slot");
        check(!store.compareExchange("retiring", "older-duty", null), "stale retirement cleanup refuses");
        check(store.compareExchange("pending", null, "browser-proof"), "browser CAS remains supported");
        check(f.stopRequests == 0 && f.writes == 0 && f.nativeWrites == 3,
            "native record CAS stays atomic without retiring the credential");
        try { store.compareExchange("unregistered", null, "value"); throw new AssertionError("unknown slot admitted"); }
        catch (IllegalArgumentException expected) { /* Only existing registered slots are admitted. */ }
    }

    private static void newerOwnerSupersedesWaitingCredentialCompare() throws Exception {
        Fixture f = fresh();
        Store older = new Store(reserve());
        f.onSleep.add(() -> {
            Store newer = new Store(reserve());
            f.confirmOnDispatch = true;
            try { check(newer.compareExchange("credential", "original", "newer"), "newer owner commits"); }
            catch (Exception failure) { throw new AssertionError(failure); }
        });
        rejects(() -> older.compareExchange("credential", "original", "older"), "Credential intent changed");
        check("newer".equals(f.value) && f.writes == 1 && f.changed == 1 && f.stopRequests == 2,
            "late older CAS cannot overwrite the newer credential");
        check(f.lease.isReleased(f.lastRequest), "older cleanup does not change newer lease release");
    }

    public static void main(String[] args) throws Exception {
        unchangedKeepsRunning();
        exactAbsenceAndReceipt();
        reservationSupersedesQueuedWrite();
        olderCleanupKeepsNewerUnconfirmedLease();
        preCommitFailureWaitsForExactReceipt(false);
        preCommitFailureWaitsForExactReceipt(true);
        writeWithLostReadbackKeepsFence();
        credentialCompareMismatchHasNoEffects();
        staleCompareChecksOwnerBeforeReading();
        credentialCompareWaitsForExactStop();
        sameCredentialCompareKeepsResident();
        credentialCompareRepairsEmptySelection();
        credentialCompareLostReadbackKeepsFence();
        nativeComparePreservesNewerReceipts();
        newerOwnerSupersedesWaitingCredentialCompare();
        System.out.println("PASS Android resident credential mutation (15 scenarios)");
    }
}
