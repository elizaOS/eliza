package ai.elizaos.app;

import ai.eliza.plugins.agent.runtime.CredentialShutdownLease;
import ai.eliza.plugins.securestore.nativeonly.CredentialIntentFence;
import ai.eliza.plugins.securestore.nativeonly.CredentialMutationConductor;
import ai.eliza.plugins.securestore.nativeonly.CredentialSlotMutation;
import java.util.ArrayDeque;

/** Runs the actual native mutation classes and the app's lifecycle binding. */
public final class ResidentCredentialMutationTest {
    /* ACTUAL_LIFECYCLE */
    /* ACTUAL_RESERVE */

    private static Fixture fixture;
    private static final class Context {}
    private static final class JsonCredentialSlots {
        static final Object LOCK = new Object();
    }
    private static final class SystemClock {
        static long elapsedRealtime() { return fixture.now; }
    }
    // Advances the fixture's clock between actual conductor polls.
    private static final class Thread {
        static void sleep(long millis) {
            fixture.now += millis;
            if (!fixture.onSleep.isEmpty()) fixture.onSleep.removeFirst().run();
        }
    }

    private static final class ElizaCredentialRetirement {
        static long reserve() { return fixture.lease.acquire(); }
        static void request(Context context, long request) {
            fixture.lastRequest = fixture.lease.retry(request);
            fixture.stopRequests++;
            if (fixture.failDispatch) throw new IllegalStateException("controlled dispatch failure");
            if (fixture.confirmOnDispatch) {
                fixture.stopped = true;
                fixture.lease.acknowledge(request, true);
            }
        }
        static boolean stopped(Context context) { return fixture.stopped; }
        static boolean acknowledged(long request) { return fixture.lease.isConfirmed(request); }
        static void release(long request) { fixture.lease.release(request); }
        static void abandon(long request) { fixture.lease.abandon(request); }
    }

    private static final class Binding {
        final Context context = new Context();
        /* ACTUAL_HOST */
    }

    private static final class Fixture {
        final CredentialShutdownLease lease = new CredentialShutdownLease();
        final ArrayDeque<Runnable> onSleep = new ArrayDeque<>();
        String value = "original";
        boolean stopped, confirmOnDispatch, failDispatch, omitWrite, failReadback;
        long now, lastRequest;
        int stopRequests, writes, changed;

        void replace(Runnable owner, String replacement) throws Exception {
            CredentialSlotMutation.replace(new Binding().host(), owner, () -> {
                if (failReadback && writes > 0) throw new IllegalStateException("controlled lost readback");
                return value;
            }, replacement, () -> {
                writes++;
                if (!omitWrite) value = replacement;
                return null;
            }, () -> {
                changed++;
                return null;
            });
        }
    }

    private static Fixture fresh() {
        check(mutationEpoch == -1, "previous mutation epoch released");
        fixture = new Fixture();
        return fixture;
    }

    private static void check(boolean condition, String message) {
        if (!condition) throw new AssertionError(message);
    }

    private interface Work { void run() throws Exception; }

    private static void rejects(Work work, String message) {
        try {
            work.run();
        } catch (IllegalStateException expected) {
            check(expected.getMessage().contains(message), "unexpected refusal: " + expected.getMessage());
            return;
        } catch (Exception failure) {
            throw new AssertionError("unexpected exception", failure);
        }
        throw new AssertionError("expected refusal: " + message);
    }

    private static void unchangedKeepsRunning() throws Exception {
        Fixture f = fresh();
        f.replace(reserve(), "original");
        check(f.stopRequests == 0 && f.changed == 0 && f.writes == 1, "same value keeps the resident");
        f.lease.beforeStart();
    }

    private static void exactAbsenceAndReceipt() throws Exception {
        Fixture f = fresh();
        f.onSleep.add(() -> {
            f.lease.acknowledge(f.lastRequest, true);
            check(f.writes == 0 && !f.stopped, "receipt alone has no durable write");
        });
        f.onSleep.add(() -> {
            check(f.writes == 0, "unconfirmed absence still blocks");
            f.lease.acknowledge(f.lastRequest, false);
            f.stopped = true;
            f.lease.acknowledge(f.lastRequest + 1, true);
            rejects(f.lease::beforeStart, "Credential shutdown is pending");
        });
        f.onSleep.add(() -> {
            check(f.writes == 0, "foreign receipt does not admit a write");
            f.lease.acknowledge(f.lastRequest, true);
        });
        f.replace(reserve(), null);
        check(f.value == null && f.changed == 1 && f.writes == 1, "sign-out writes once after both checks");
        check(f.lease.isReleased(f.lastRequest), "confirmed readback releases exact lease");
        f.lease.beforeStart();
    }

    private static void reservationSupersedesQueuedWrite() throws Exception {
        Fixture f = fresh();
        Runnable earlier = reserve();
        Runnable current = reserve();
        rejects(() -> f.replace(earlier, "older"), "Credential intent changed");
        check(f.writes == 0 && f.stopRequests == 0, "stale queued owner has no effects");
        f.confirmOnDispatch = true;
        f.replace(current, "newer");
        check("newer".equals(f.value) && f.writes == 1, "current owner writes");
    }

    private static void olderCleanupKeepsNewerUnconfirmedLease() throws Exception {
        Fixture f = fresh();
        f.onSleep.add(() -> {
            Runnable newer = reserve();
            f.confirmOnDispatch = true;
            f.omitWrite = true;
            rejects(() -> f.replace(newer, "newer"), "Credential commit unconfirmed");
            check(f.lease.isConfirmed(f.lastRequest), "newer exact stop is confirmed");
        });
        rejects(() -> f.replace(reserve(), "older"), "Credential intent changed");
        check(f.stopRequests == 2 && f.writes == 1 && "original".equals(f.value), "only newer write attempted");
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
    }

    private static void preCommitFailureWaitsForExactReceipt(boolean dispatch) {
        Fixture f = fresh();
        f.failDispatch = dispatch;
        f.onSleep.add(() -> f.now = 15_000);
        rejects(() -> f.replace(reserve(), "newer"),
            dispatch ? "controlled dispatch failure" : "Native shutdown unconfirmed");
        check(f.writes == 0 && f.changed == 0 && "original".equals(f.value), "pre-commit failure has no write");
        f.lease.acknowledge(f.lastRequest + 1, true);
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
        f.lease.acknowledge(f.lastRequest, true);
        check(f.lease.isReleased(f.lastRequest), "late exact receipt releases an abandoned pre-commit stop");
        f.lease.beforeStart();
    }

    private static void writeWithLostReadbackKeepsFence() {
        Fixture f = fresh();
        f.confirmOnDispatch = true;
        f.failReadback = true;
        rejects(() -> f.replace(reserve(), "newer"), "controlled lost readback");
        check("newer".equals(f.value) && f.changed == 1 && f.writes == 1, "write may have happened");
        check(f.lease.isConfirmed(f.lastRequest) && !f.lease.isReleased(f.lastRequest), "unconfirmed write keeps exact stop");
        rejects(f.lease::beforeStart, "Credential shutdown is pending");
    }

    public static void main(String[] args) throws Exception {
        unchangedKeepsRunning();
        exactAbsenceAndReceipt();
        reservationSupersedesQueuedWrite();
        olderCleanupKeepsNewerUnconfirmedLease();
        preCommitFailureWaitsForExactReceipt(false);
        preCommitFailureWaitsForExactReceipt(true);
        writeWithLostReadbackKeepsFence();
        System.out.println("PASS Android resident credential mutation (7 scenarios)");
    }
}
