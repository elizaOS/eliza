 /** Rotation: the key a renewal replaced is revoked even when the process ends before Cloud confirmed it. */
 static void retirementScenarios()throws Exception{
  // 13. Cloud does not answer the revocation: the duty stays stored, is retried in this process, and again on a later start.
  CloudSignInChainTest offline=renewalHost(5*DAY,"ok");CloudKeyRevocation.unanswered=true;

  check("renewed".equals(renewed.result.getString("status"))&&NEW.equals(savedToken())&&revoked.isEmpty()&&CloudKeyRevocation.retirements.equals(List.of(OLD)),"the new key is saved although the old one could not be revoked yet");
  JSONObject duty=new JSONObject(slots.get(RETIRING));
  check(OLD.equals(duty.getString("token"))&&duty.length()==2&&duty.getLong("expiresAt")>System.currentTimeMillis(),"the duty names the replaced key and when it lapses on its own");
  check(offline.main.delays.equals(List.of(5000L)),"one retry is scheduled in this process");
  offline.main.delays.remove(0);offline.main.delayed.remove().run();offline.drainRenewal();
  check(revoked.isEmpty()&&CloudKeyRevocation.retirements.size()==2&&slots.containsKey(RETIRING)&&offline.main.delays.equals(List.of(25000L)),"an unanswered retry keeps the duty");
  // The process ends here. The next one finds the duty and no closure.
  CloudSignInChainTest next=new CloudSignInChainTest();next.observable=false;
  next.renewSaved();check(revoked.isEmpty(),"nothing is sent on the main thread");next.drainRenewal();
  check(revoked.isEmpty()&&slots.containsKey(RETIRING),"still unanswered: the duty waits");
  CloudKeyRevocation.unanswered=false;
  CloudSignInChainTest later=new CloudSignInChainTest();later.observable=false;
  later.renewSaved();later.drainRenewal();
  check(revoked.equals(List.of(OLD))&&!slots.containsKey(RETIRING)&&NEW.equals(savedToken()),"a later start revokes the replaced key from storage, whether or not the host still renews, and forgets the duty");
  check(later.requests==0&&later.reserved==0&&notifications.size()==1,"it sends no sign-in request, reserves nothing and announces nothing");
  // 14. The process ends between the stored duty and the save: the saved, working key is never revoked.
  CloudSignInChainTest interrupted=renewalHost(5*DAY,"ok");interrupted.dieBeforeKeyWrite=true;

  check(revoked.equals(List.of(NEW))&&CloudKeyRevocation.retirements.isEmpty(),"only the unsaved new key is revoked");

  check(!slots.containsKey(RETIRING)&&CloudKeyRevocation.retirements.isEmpty()&&OLD.equals(savedToken())&&restarted.requests==0,"a duty for the key that is still saved is dropped without a request");

  slots.put(RETIRING,new JSONObject().put("token","eliza_mobile_"+"d".repeat(64)).put("expiresAt",System.currentTimeMillis()-1000).toString());
  CloudSignInChainTest lapsed=new CloudSignInChainTest();lapsed.renewSaved();lapsed.drainRenewal();
  check(!slots.containsKey(RETIRING)&&CloudKeyRevocation.retirements.isEmpty(),"an expired replaced key is forgotten");
  // 15. A renewal that cannot store the duty saves nothing: it never forks the sign-in.
  CloudSignInChainTest unstored=renewalHost(5*DAY,"ok");unstored.failRetiringWrite=true;
  check("deferred".equals(unstored.renewNowCall().result.getString("status")),"without the stored duty the renewal fails");

  check("renewed".equals(hostFails.renewNowCall().result.getString("status"))&&revoked.equals(List.of(OLD))&&notifications.size()==1&&events.indexOf("readmit")<events.indexOf("revoke"),"a failed host re-admission still revokes the old key afterwards and tells the renderer");

 static CloudSignInChainTest recoverReceipt(String input, String expect, String... replies)throws Exception{
  slots.clear();events.clear();notifications.clear();revoked.clear();
  CloudSignInChainTest host=new CloudSignInChainTest();host.mode="recovery";
  host.recoveryReplies.addAll(List.of(replies));
  String receiptSlot=host.config().receiptSlot(TestRegistration.PRODUCTION);
  String savedSlot=host.config().credentialSlot(TestRegistration.PRODUCTION);
  String verifier=CloudSignInFlow.newVerifier();
  host.issuedState=CloudSignInFlow.newState();
  host.issuedChallenge=CloudSignInFlow.s256(verifier);
  JSONObject receipt=new JSONObject().put("clientId","ai.example.signin")
   .put("environment",input.equals("wrong-environment")?"staging":"production")
   .put("redirectUri",TestRegistration.PRODUCTION.callback())
   .put("code","emac_"+"a".repeat(64)).put("state",host.issuedState)
   .put("codeVerifier",verifier).put("credentialId",ID)
   .put("acknowledgeBy",input.startsWith("expired")?1:System.currentTimeMillis()+60000);
  if(!input.equals("absent"))slots.put(receiptSlot,input.equals("malformed")?"invalid-fixture-frame":receipt.toString());
  if(!input.equals("missing-key"))slots.put(savedSlot,new JSONObject().put("credentialId",REQUEST)
   .put("serverCredentialId",input.equals("wrong-credential")?REQUEST:ID)
   .put("token",NEW).put("expiresAt",System.currentTimeMillis()+60000).toString());
  slots.put("cloud:staging","preserved-other-environment");
  host.failReceiptRead=input.equals("read-failure");
  host.failReceiptRemove=input.equals("remove-failure");
  host.destroyed=input.equals("destroyed");
  host.observable=!input.endsWith("-unobserved");
  host.receiptBefore=slots.get(receiptSlot);host.keyBefore=slots.get(savedSlot);
  host.recover();
  // The request after a passed deadline waits for its turn on the main thread, as the in-process rounds do.
  if(input.startsWith("expired")){
   check(host.requests==0&&host.activating.contains("production")&&host.main.delays.equals(List.of(1000L)),"a passed deadline schedules the one request after it: "+input);
   host.nextAcknowledgement();
  }
  check(host.requests==replies.length && host.recoveryReplies.isEmpty(),"recovery round count: "+input);
  host.recovered(expect,input);
  return host;
 }
 /**
  * `stale`: the receipt names no saved key and goes. `kept`: nothing is settled yet. `confirmed`: the
  * key stays, the receipt goes and the sign-in is announced. `expired`: the dead key and its receipt
  * are removed, the key is revoked and the owner is told. `closed`: the plugin closed meanwhile.
  */
 void recovered(String expect,String input){
  String receiptSlot=config().receiptSlot(TestRegistration.PRODUCTION), savedSlot=config().credentialSlot(TestRegistration.PRODUCTION);
  check("preserved-other-environment".equals(slots.get("cloud:staging"))&&reserved==0,"startup recovery touches no other deployment and creates no interactive attempt: "+input);
  switch(expect){
   case "stale" -> check(!slots.containsKey(receiptSlot)&&Objects.equals(slots.get(savedSlot),keyBefore)&&notifications.isEmpty()&&revoked.isEmpty()&&!activating.contains("production")&&main.delayed.isEmpty(),"a receipt for no saved key is dropped and nothing else changes: "+input);
   case "kept","closed" -> {
    check(Objects.equals(slots.get(receiptSlot),receiptBefore)&&Objects.equals(slots.get(savedSlot),keyBefore)&&notifications.isEmpty()&&revoked.isEmpty(),"nothing is settled: "+input);
    if(expect.equals("closed"))check(main.delayed.isEmpty(),"a closed plugin schedules nothing: "+input);
   }
   case "confirmed" -> {
    check(Objects.equals(slots.get(savedSlot),keyBefore)&&revoked.isEmpty()&&!activating.contains("production")&&main.delayed.isEmpty(),"a confirmed key stays saved: "+input);
    if("newer-receipt".equals(slots.get(receiptSlot)))check(!"newer-receipt".equals(receiptBefore),"a receipt a newer sign-in wrote stays: "+input);
    else if(input.equals("remove-failure"))check(Objects.equals(slots.get(receiptSlot),receiptBefore),"a receipt that cannot be removed waits for the next start: "+input);
    else check(!slots.containsKey(receiptSlot),"its receipt goes: "+input);
    check(notifications.size()==1&&events.contains("signInCompleted"),"the sign-in is announced once, retained: "+input);
    JSObject outcome=notifications.get(0);
    check("authenticated".equals(outcome.getString("status"))&&"production".equals(outcome.getString("environment"))&&outcome.length()==4&&outcome.has("expiresAt")&&outcome.opt("accountCreated")==JSONObject.NULL&&!outcome.toString().contains(NEW),"the unscoped four-field outcome, without the key: "+input);
   }
   case "expired" -> {
    check(!slots.containsKey(savedSlot)&&!slots.containsKey(receiptSlot)&&revoked.equals(List.of(NEW)),"the key Cloud will never accept is removed with its receipt and revoked: "+input);
    check(notifications.size()==1&&events.contains("signInIncomplete")&&"activation-expired".equals(notifications.get(0).getString("code"))&&notifications.get(0).length()==1,"the owner is told once with the fixed code: "+input);
    check(!activating.contains("production")&&main.delayed.isEmpty(),"nothing further is pending: "+input);
   }
   default -> throw new AssertionError(expect);
  }
 }

  recoverReceipt("expired","expired","retry");
  recoverReceipt("expired","expired","denied");

  recoverReceipt("expired","confirmed","complete");
  recoverReceipt("expired","confirmed","success");

   CloudSignInChainTest waiting=recoverReceipt("current","kept",first);
   check(waiting.activating.contains("production")&&waiting.main.delays.equals(List.of(5000L)),"an unanswered startup acknowledgement is pending and asked again: "+first);
   waiting.recoveryReplies.add("retry");waiting.nextAcknowledgement();
   check(waiting.main.delays.equals(List.of(10000L))&&notifications.isEmpty()&&waiting.activating.contains("production"),"asked again with a longer wait: "+first);
   waiting.recoveryReplies.add(first.equals("wrong")?"success":"denied");waiting.nextAcknowledgement();
   waiting.recovered(first.equals("wrong")?"confirmed":"expired","restart between background rounds, "+first);
  }
  CloudSignInChainTest exhausted=recoverReceipt("current","kept","timeout","rate","retry");
  check(exhausted.activating.contains("production")&&exhausted.main.delayed.size()==1,"three unanswered startup rounds hand over to the background rounds");

  restart.recoveryReplies.add("success");
  String saved=slots.get(restart.config().credentialSlot(TestRegistration.PRODUCTION));
  restart.recover();
  check(restart.requests==1 && restart.recoveryReplies.isEmpty(),"new startup retries the saved receipt");
  check(!slots.containsKey(receiptSlot),"new startup removes the acknowledged receipt");
  check(saved.equals(slots.get(restart.config().credentialSlot(TestRegistration.PRODUCTION)))&&notifications.size()==1&&events.contains("signInCompleted"),"new startup preserves the saved key and announces it");
  // A newer credential action after the restart owns the slot: the old recovery settles nothing.
  CloudSignInChainTest older=recoverReceipt("current","kept","wrong");
  older.intent++;older.nextAcknowledgement();
  check(older.requests==1&&!older.activating.contains("production")&&older.main.delayed.isEmpty(),"a superseded recovery sends nothing more");
  older.recovered("kept","superseded after restart");
  CloudSignInChainTest lateOlder=recoverReceipt("current","kept","wrong");
  lateOlder.recoveryReplies.add("denied");
  String replacement=new JSONObject().put("credentialId",SAVED_ID).put("serverCredentialId",SAVED_ID).put("token",OLD).put("expiresAt",System.currentTimeMillis()+60000).toString();
  slots.put("cloud:production",replacement);lateOlder.nextAcknowledgement();
  check(replacement.equals(slots.get("cloud:production"))&&!events.contains("signInCompleted"),"a key saved since the restart is never removed for the old receipt");
  // A host that cannot observe its credential intent still settles by the exact saved key.
  check(lateOlder.observations==1,"the intent is observed once, before the read");
  CloudSignInChainTest plain=recoverReceipt("expired-unobserved","expired","retry");
  check(plain.observations==0,"without an observable intent the exact saved key alone bounds the removal");
 }
