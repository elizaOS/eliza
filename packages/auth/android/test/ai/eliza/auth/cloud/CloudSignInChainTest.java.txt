 /** SMS sends share the email cooldown and honour Cloud's longer wait (Retry-After or resendAt). */
 static void phoneCooldownFollowsCloud()throws Exception{
  slots.clear();events.clear();notifications.clear();
  CloudSignInChainTest host=new CloudSignInChainTest();host.mode="sms-limited";
  long before=System.currentTimeMillis();
  PluginCall limited=phoneCall();host.startPhone(limited);
  check("rate-limited".equals(limited.error)&&host.attempt==null,"a refused SMS send is explicit");
  check(host.emailSendAllowedAt>=before+120000&&host.emailSendAllowedAt<=System.currentTimeMillis()+120000,"Retry-After on a refused SMS send sets the shared wait");
  host.mode="valid";host.lastEmailSendAt=0;
  PluginCall early=phoneCall();host.startPhone(early);
  check("cooldown".equals(early.error)&&host.requests==1,"an SMS send inside Cloud's wait sends nothing, although the local 30 seconds have passed");
  PluginCall emailEarly=sendCall();host.startEmail(emailEarly);
  check("cooldown".equals(emailEarly.error)&&host.requests==1,"the wait applies to an email send as well");
  host.emailSendAllowedAt=0;host.smsResendAt=Instant.ofEpochMilli(System.currentTimeMillis()+90000).toString();
  long resend=Instant.parse(host.smsResendAt).toEpochMilli();
  PluginCall sent=phoneCall();host.startPhone(sent);
  check(sent.result!=null&&host.attempt!=null&&host.attempt.phone!=null,"SMS code sent");
  check(sent.result.getLong("resendAt")==resend&&host.emailSendAllowedAt==resend,"the code step reports Cloud's resend time, not the local 30 seconds");
  host.lastEmailSendAt=0;
  PluginCall again=phoneCall();host.startPhone(again);
  check("cooldown".equals(again.error)&&host.requests==2,"a resend waits for Cloud's time");
  host.emailSendAllowedAt=0;host.smsResendAt=null;long now=System.currentTimeMillis();
  PluginCall plain=phoneCall();host.startPhone(plain);
  check(plain.result!=null&&plain.result.getLong("resendAt")>=now+CloudSignInFlow.RESEND_COOLDOWN_MS,"without a Cloud time the local minimum still applies");
 }
