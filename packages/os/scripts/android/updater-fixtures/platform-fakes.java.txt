    public java.net.URLConnection openConnection(java.net.URL url) throws java.io.IOException {
        android.Device.fetched.add(url.toString());
        byte[] body = android.Device.channel.get(url.toString());
        if (body == null) throw new java.io.FileNotFoundException(url.toString());
        return new java.net.URLConnection(url) {
            @Override public void connect() {}
            @Override public java.io.InputStream getInputStream() { return new java.io.ByteArrayInputStream(body); }
        };
    }
}

package android.os;
public class UpdateEngine {
    public static final class UpdateStatusConstants { public static final int IDLE = 0, DOWNLOADING = 3, VERIFYING = 4, FINALIZING = 5, UPDATED_NEED_REBOOT = 6; }
    public static final class ErrorCodeConstants { public static final int SUCCESS = 0, ERROR = 1; }
    public static final java.util.List<Object[]> applied = new java.util.ArrayList<>();
    public static int status, unbinds;
    public static boolean denied;
    public static RuntimeException applyFailure;
    /** The work's result, delivered as soon as there is work; null leaves the test to deliver it. */
    public static Integer outcome;
    public static volatile UpdateEngineCallback callback;
    public UpdateEngine() { if (denied) throw new IllegalStateException("Failed to find update_engine"); }
    public boolean bind(UpdateEngineCallback bound) {
        bound.onStatusUpdate(status, 0f);
        if (status != UpdateStatusConstants.IDLE && outcome != null) bound.onPayloadApplicationComplete(outcome);
        callback = bound;
        return true;
    }
    public boolean unbind() { unbinds++; return true; }
    public void applyPayload(String url, long offset, long size, String[] headers) {
        if (applyFailure != null) throw applyFailure;
        applied.add(new Object[] {url, offset, size, headers});
        if (outcome != null) callback.onPayloadApplicationComplete(outcome);
    }
}

// file: org/json/JSONObject.java
package org.json;
public class JSONObject {
    private final java.util.Map<String, Object> values = new java.util.HashMap<>();
    private String text;
    private int at;
    private JSONObject() {}
    public JSONObject(String source) throws JSONException {
        text = source;
        if (space() != '{') throw new JSONException("Expected an object");
        JSONObject parsed = (JSONObject) value();
        if (at != text.length() && space() != 0) throw new JSONException("Trailing data");
        values.putAll(parsed.values);
    }
    static <T> T typed(Object value, Class<T> type) throws JSONException {
        if (!type.isInstance(value)) throw new JSONException("Expected " + type.getSimpleName());
        return type.cast(value);
    }
    public boolean isNull(String key) { return values.get(key) == null; }
    public String getString(String key) throws JSONException { return typed(values.get(key), String.class); }
    public long getLong(String key) throws JSONException { return typed(values.get(key), Long.class); }
    public int getInt(String key) throws JSONException { return (int) getLong(key); }
    public JSONObject getJSONObject(String key) throws JSONException { return typed(values.get(key), JSONObject.class); }
    public JSONArray getJSONArray(String key) throws JSONException { return typed(values.get(key), JSONArray.class); }
    private char space() { while (at < text.length() && Character.isWhitespace(text.charAt(at))) at++; return at < text.length() ? text.charAt(at) : 0; }
    private void expect(char wanted) throws JSONException { if (space() != wanted) throw new JSONException("Expected " + wanted + " at " + at); at++; }
    private Object value() throws JSONException {
        char first = space();
        if (first == '{') {
            at++;
            JSONObject object = new JSONObject();
            while (space() != '}') {
                if (!object.values.isEmpty()) expect(',');
                String key = (String) value();
                expect(':');
                object.values.put(key, value());
            }
            at++;
            return object;
        }
        if (first == '[') {
            at++;
            JSONArray array = new JSONArray();
            while (space() != ']') {
                if (array.length() > 0) expect(',');
                array.values.add(value());
            }
            at++;
            return array;
        }
        if (first == '"') {
            int end = text.indexOf('"', at + 1);
            if (end < 0) throw new JSONException("Unterminated string");
            String value = text.substring(at + 1, end);
            at = end + 1;
            return value;
        }
        int start = at;
        while (at < text.length() && ",]} \n".indexOf(text.charAt(at)) < 0) at++;
        String token = text.substring(start, at);
        if (token.equals("null")) return null;
        try { return Long.valueOf(token); } catch (NumberFormatException failure) { throw new JSONException("Bad token " + token); }
    }
}
