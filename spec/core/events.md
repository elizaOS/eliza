# Agent events and subscriptions

Event sequence, session association, callback identity and failure reporting are runtime-owned. Sources include [agent-event.ts](https://github.com/elizaOS/eliza/blob/72fed7e4069ecd35d61d886ab6eec5c02c137517/packages/core/src/services/agent-event.ts), [agent-event-bridge.ts](https://github.com/elizaOS/eliza/blob/72fed7e4069ecd35d61d886ab6eec5c02c137517/packages/core/src/services/agent-event-bridge.ts) and [AgentEventService.qnt](https://github.com/elizaOS/eliza/blob/72fed7e4069ecd35d61d886ab6eec5c02c137517/packages/core/quint/AgentEventService.qnt).

## Story: Emit ordered events with their run session

As a host I receive each original event before diagnostics generated by its listeners, with independent sequence counters for every run.

### Original delivery precedes its failure diagnostics

Each run has a monotonic sequence until that run is cleared. Explicit caller session takes precedence over its saved run session.

```quint
type EventPhase = EventIdle | DeliveringOriginal | ReportingListenerError | EmittingWarning
type HeartbeatStatus = NoHeartbeat | HeartbeatSent | HeartbeatFailed
type HeartbeatIndicator = NoIndicator | AlertIndicator | ErrorIndicator
type RunClock = { id: str, sequence: int, session: str }
type Subscription = { callback: int, channel: str, generation: int }
type EventState = { runs: List[RunClock], phase: EventPhase, activeRun: str, baseSequence: int, eventSequence: int, requestedSession: str, savedSession: str, eventSession: str, received: List[int], listeners: List[Subscription], nextGeneration: int, pending: List[Subscription], delivered: List[Subscription], listenerFailed: bool, heartbeat: HeartbeatStatus, indicator: HeartbeatIndicator }
pure val initial: EventState = { runs: List(), phase: EventIdle, activeRun: "", baseSequence: 0, eventSequence: 0, requestedSession: "", savedSession: "", eventSession: "", received: List(), listeners: List(), nextGeneration: 1, pending: List(), delivered: List(), listenerFailed: false, heartbeat: NoHeartbeat, indicator: NoIndicator }
pure val runIdValues = Set("run-a", "run-b")
pure val sessionValues = Set("", "saved-session", "caller-session")
pure val callbackValues = Set(1, 2, 3)
pure val channelValues = Set("event", "heartbeat")
pure val generationValues = Set(1, 2, 3, 4)
pure val missingRun: RunClock = {id: "", sequence: 0, session: ""}
pure def eventRun(s: EventState, runId: str): RunClock = s.runs.select(r => r.id == runId).nthOr(0, missingRun)
pure def canRegisterRun(s: EventState, runId: str, session: str): bool = s.phase == EventIdle and runId != ""
pure def registerRun(s: EventState, runId: str, session: str): EventState = { ...s, runs: if (eventRun(s, runId).id == "") s.runs.append({id: runId, sequence: 0, session: session}) else s.runs.listMap(r => if (r.id == runId) { ...r, session: session } else r) }
pure def canClearRun(s: EventState, runId: str): bool = s.phase == EventIdle
pure def clearRun(s: EventState, runId: str): EventState = { ...s, runs: s.runs.select(r => r.id != runId) }
pure def canAdmitEvent(s: EventState, runId: str, session: str): bool = s.phase == EventIdle and eventRun(s, runId).id != ""
pure def admitEvent(s: EventState, runId: str, session: str): EventState = {
  val saved = eventRun(s, runId)
  { ...s, phase: DeliveringOriginal, activeRun: runId, baseSequence: saved.sequence, eventSequence: saved.sequence + 1,
    runs: s.runs.listMap(r => if (r.id == runId) { ...r, sequence: r.sequence + 1 } else r), requestedSession: session, savedSession: saved.session, eventSession: if (session == "") saved.session else session,
    received: List(), pending: s.listeners.select(h => h.channel == "event"), delivered: List(), listenerFailed: false }
}
pure def canFinishOriginal(s: EventState): bool = s.phase == DeliveringOriginal and s.pending.length() == 0
pure def finishOriginal(s: EventState): EventState = { ...s, received: s.received.append(s.eventSequence), phase: if (s.listenerFailed) ReportingListenerError else EventIdle }
pure def canReportListenerError(s: EventState): bool = s.phase == ReportingListenerError
pure def reportListenerError(s: EventState): EventState = { ...s, phase: EmittingWarning, runs: s.runs.listMap(r => if (r.id == s.activeRun) { ...r, sequence: r.sequence + 1 } else r), received: s.received.append(s.eventSequence + 1) }
pure def canEmitListenerWarning(s: EventState): bool = s.phase == EmittingWarning
pure def emitListenerWarning(s: EventState): EventState = { ...s, phase: EventIdle, runs: s.runs.listMap(r => if (r.id == s.activeRun) { ...r, sequence: r.sequence + 1 } else r), received: s.received.append(s.eventSequence + 2) }
pure def alwaysRunSequencesNonnegative(s: EventState): bool = s.runs.every(r => r.sequence >= 0 and s.runs.select(x => x.id == r.id).length() == 1)
pure def alwaysSessionUsesCallerOrRun(s: EventState): bool = s.eventSession == (if (s.requestedSession == "") s.savedSession else s.requestedSession)
pure def alwaysDiagnosticsFollowOriginal(s: EventState): bool = (s.phase == DeliveringOriginal implies s.received.length() == 0) and (s.phase == ReportingListenerError implies s.received == List(s.eventSequence)) and (s.phase == EmittingWarning implies s.received == List(s.eventSequence, s.eventSequence + 1))
type EventSequenceCursor = {expected: int, valid: bool}
pure def sequenceCursor(expected: int): EventSequenceCursor = {expected: expected, valid: true}
pure def advanceSequenceCursor(acc: EventSequenceCursor, n: int): EventSequenceCursor = {expected: acc.expected + 1, valid: acc.valid and n == acc.expected}
pure def alwaysReceivedInSequence(s: EventState): bool = s.received.foldl(sequenceCursor(s.baseSequence + 1), (acc, n) => advanceSequenceCursor(acc, n)).valid
pure def sometimesErrorAndWarningFollowOriginal(s: EventState): bool = s.phase == EventIdle and s.listenerFailed and s.received.length() == 3
```

## Story: Retire the exact subscription handle

As a transport I detach a stale handle without removing a replacement, and one callback detaching itself never skips its peers.

### Each callback has one current generation per channel

Duplicate subscription preserves the live handle. Detach only removes a matching generation.

```quint
pure def canSubscribe(s: EventState, callback: int, channel: str): bool = callback > 0 and channel != ""
pure def subscribe(s: EventState, callback: int, channel: str): EventState =
  if (s.listeners.some(h => h.callback == callback and h.channel == channel)) s
  else { ...s, listeners: s.listeners.append({callback: callback, channel: channel, generation: s.nextGeneration}), nextGeneration: s.nextGeneration + 1 }
pure def canDetach(s: EventState, callback: int, channel: str, generation: int): bool = callback > 0 and channel != "" and generation > 0
pure def detach(s: EventState, callback: int, channel: str, generation: int): EventState = { ...s, listeners: s.listeners.select(h => h.callback != callback or h.channel != channel or h.generation != generation) }
pure def canDeliverListener(s: EventState, ok: bool, detachSelf: bool): bool = s.phase == DeliveringOriginal and s.pending.length() > 0
pure def deliverListener(s: EventState, ok: bool, detachSelf: bool): EventState = {
  val listener = s.pending.nthOr(0, {callback: 0, channel: "", generation: 0})
  { ...s, pending: s.pending.removeAt(0), delivered: s.delivered.append(listener), listenerFailed: s.listenerFailed or not(ok),
    listeners: if (detachSelf) s.listeners.select(h => h.callback != listener.callback or h.channel != listener.channel or h.generation != listener.generation) else s.listeners }
}
pure def alwaysOneCurrentRegistration(s: EventState): bool = s.listeners.every(h => h.generation > 0 and h.generation < s.nextGeneration and s.listeners.select(x => x.callback == h.callback and x.channel == h.channel).length() == 1)
pure def alwaysSnapshotCallbacksUnique(s: EventState): bool = s.pending.every(h => not(s.delivered.some(d => d.generation == h.generation)))
pure def sometimesStaleDetachPreservesReplacement(s: EventState): bool = s.listeners.some(h => h.generation > 1)
pure def sometimesDetachedCallbackLeavesPeerPending(s: EventState): bool = s.delivered.some(d => not(s.listeners.some(h => h.generation == d.generation))) and s.pending.length() > 0
```

### Heartbeat status determines its indicator and stop clears subscriptions

Stopping clears run clocks, active registrations and heartbeat state when synchronous fanout has finished.

```quint
pure def canHeartbeat(s: EventState, status: HeartbeatStatus): bool = status != NoHeartbeat
pure def heartbeat(s: EventState, status: HeartbeatStatus): EventState = { ...s, heartbeat: status, indicator: if (status == HeartbeatFailed) ErrorIndicator else AlertIndicator }
pure def canStopEvents(s: EventState): bool = s.phase == EventIdle
pure def stopEvents(s: EventState): EventState = { ...s, runs: List(), listeners: List(), heartbeat: NoHeartbeat, indicator: NoIndicator }
pure def alwaysHeartbeatIndicatorAgrees(s: EventState): bool = (s.heartbeat == NoHeartbeat and s.indicator == NoIndicator) or (s.heartbeat == HeartbeatSent and s.indicator == AlertIndicator) or (s.heartbeat == HeartbeatFailed and s.indicator == ErrorIndicator)
pure def sometimesFailedHeartbeatVisible(s: EventState): bool = s.heartbeat == HeartbeatFailed and s.indicator == ErrorIndicator
```

### An admitted event eventually finishes its listeners and diagnostics

Each synchronous listener finishes or throws. The original event then precedes any resulting diagnostics and the service becomes idle.

```quint
pure val fair_deliverListener = "weak"
pure val fair_finishOriginal = "weak"
pure val fair_reportListenerError = "weak"
pure val fair_emitListenerWarning = "weak"
pure def leadsToEventIdle(s: EventState, later: EventState): bool = s.phase != EventIdle implies later.phase == EventIdle
```

### Stopping clears run and subscription indexes together

The stop transition clears the active indexes and heartbeat without changing unrelated source evidence.

```quint
pure def stepEventStopClearsIndexes(before: EventState, via: str, after: EventState): bool = via == "stopEvents" implies after.runs.length() == 0 and after.listeners.length() == 0 and after.heartbeat == NoHeartbeat and after.indicator == NoIndicator
pure def lawStaleHandleKeepsRegistrations(s: EventState, callback: int, channel: str, generation: int): bool = not(s.listeners.some(h => h.callback == callback and h.channel == channel and h.generation == generation)) implies detach(s, callback, channel, generation).listeners == s.listeners
```
